		}
	}

		# add modifiers
		bpm_underperforming_economy
	}

	is_shown_when_inactive = {
		# this should always be yes
		always = yes
	}

	possible = {
		scope:journal_entry = {
			NOT = {
				has_variable = bpm_crisis_ended
			}
		}
		# starting triggers
		any_country = {
			country_rank >= rank_value:great_power
			has_war_with = ROOT
		}
	}

	invalid = {
		hidden_trigger = {
			always = no
		}

			}
		}
